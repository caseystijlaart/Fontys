= UC_002 : Turn random direction


Scenario 1::
The ultrasonic sensor measures that an object is within a 10cm range from the robot. Therefor the robot will randomly turn in a direction. <<seq-uc_002-scenario-1>>.


[[seq-uc_002-scenario-1]]
.Turn random direction
[plantuml,format=svg]
----
@startuml

actor "Robot" as robot
participant "Wheel 1" as wheel1
participant "Wheel 2" as wheel2
participant "Ultrasonic" as sensor

robot -> sensor : HCSR04_Read()
activate sensor
activate robot
sensor -> robot : Distance
deactivate sensor

robot -> wheel1 : turnRandom(wheel1, wheel2, float curSpeed)
activate wheel1

robot -> wheel2 : turnRandom(wheel1, wheel2, float curSpeed)
activate wheel2

deactivate robot

wheel1 -> wheel1 : rotate (wheel1, float curSpeed)
deactivate wheel1
wheel2 -> wheel2 : rotate (wheel2, float curSpeed)
deactivate wheel2

@enduml
